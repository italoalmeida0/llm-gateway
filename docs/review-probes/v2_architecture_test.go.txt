// Diagnostic probes for the V2 release review; see docs/review-probes/README.md.
// Assertions describe the intended behavior, not the observed bug.
func TestV2ArchitectureHealthySilentProvider(t *testing.T) {
	previous := tuneWatchEvery
	tuneWatchEvery = 50 * time.Millisecond
	defer func() { tuneWatchEvery = previous }()
	for _, watchdog := range []bool{false, true} {
		name := "without_watchdog"
		if watchdog {
			name = "with_watchdog"
		}
		t.Run(name, func(t *testing.T) {
			started := make(chan struct{})
			finished := make(chan bool, 1)
			srv := reviewGateway(t, func(w http.ResponseWriter, r *http.Request) {
				w.Header().Set("Content-Type", "text/event-stream")
				w.(http.Flusher).Flush()
				close(started)
				select {
				case <-r.Context().Done():
					finished <- false
				case <-time.After(450 * time.Millisecond):
					reviewTextSSE(w)
					finished <- true
				}
			})
			a := reviewActor(t)
			a.supCfg.store(&DaemonConfig{HostID: "h", GatewayURL: srv.URL, Settings: HarnessSettings{NoAutoTitle: true}})
			a.rec.Options = SessionOptions{Access: "full", Mode: "talk", Effort: "none"}
			sup := newSessionSupervisor(a.store.dataDir, a.supCfg, nil, nil)
			sup.resident[a.id] = &residentEntry{handle: &sessionHandle{inbox: a.inbox, control: a.control, done: a.done}}
			go a.run()
			defer func() { a.control <- shutdownMsg{}; <-a.done }()
			reply := make(chan any, 1)
			a.inbox <- Envelope{Payload: userPromptMsg{Text: "Respond when ready", Reply: reply}}
			if result := reviewRecv(t, reply).(promptResult); !result.Accepted {
				t.Fatal(result)
			}
			reviewRecv(t, started)
			time.Sleep(150 * time.Millisecond)
			if watchdog {
				sup.watchdogRound()
			}
			if !reviewRecv(t, finished) {
				t.Error("watchdog cancelled a healthy HTTP request while the provider was still preparing its response")
			}
		})
	}
}

func TestV2ArchitectureCriticalWSDelivery(t *testing.T) {
	w := newWSActor()
	blocked, release := make(chan struct{}), make(chan struct{})
	var received atomic.Int64
	var requests atomic.Int64
	w.send = func(ev any) error {
		if received.Load() == 0 {
			close(blocked)
			<-release
		}
		if ev.(map[string]any)["type"] == "tool_approval_request" {
			requests.Add(1)
		}
		received.Add(1)
		return nil
	}
	var wg sync.WaitGroup
	wg.Add(1)
	go w.run(&wg)
	defer func() { w.control <- shutdownMsg{}; wg.Wait() }()
	w.emit(map[string]any{"type": "agent_event"})
	reviewRecv(t, blocked)
	for i := 0; i < cap(w.outbound); i++ {
		w.emit(map[string]any{"type": "agent_event"})
	}
	a := reviewActor(t)
	a.state, a.gen = stateRunning, 1
	a.rec.Options = SessionOptions{Access: "ask", Mode: "build"}
	a.wsSend = func(ev any) { w.emit(ev) }
	a.onWorkerApprovalReq(workerApprovalReqMsg{gen: 1, id: "approval", callID: "approval", tool: "bash", reply: make(chan approvalOutcome, 1)})
	defer a.clearPending()
	close(release)
	deadline := time.Now().Add(time.Second)
	for received.Load() < int64(cap(w.outbound)+1) && time.Now().Before(deadline) {
		time.Sleep(time.Millisecond)
	}
	if requests.Load() != 1 {
		t.Errorf("approval request lost: actor state=%s, delivered=%d, dropped=%d", a.state, requests.Load(), w.dropped.Load())
	}
}

func TestV2ArchitectureBGCompletionDelivery(t *testing.T) {
	b := newBGSupervisor(t.TempDir())
	inbox := make(chan Envelope, 1)
	inbox <- Envelope{Payload: workerHeartbeatMsg{gen: 1}}
	b.session = func(string) (chan Envelope, chan any, bool) { return inbox, nil, true }
	reply := make(chan any, 1)
	b.onRegister(bgRegisterMsg{SessionID: "session", Kind: "bash", Label: "build", Reply: reply})
	jobID := (<-reply).(bgRegisterResult).JobID
	b.onFinish(jobID, BgStatusDone, "done")
	<-inbox
	select {
	case env := <-inbox:
		if _, ok := env.Payload.(bgNoticeMsg); !ok {
			t.Fatalf("unexpected notice: %T", env.Payload)
		}
	default:
		t.Error("background completion permanently discarded when the session inbox is full")
	}
}

func TestV2ArchitectureDispatchIsolation(t *testing.T) {
	a := reviewActor(t)
	stalled, release := make(chan struct{}), make(chan struct{})
	a.inbox <- Envelope{Payload: hookMsg{fn: func() { close(stalled); <-release }}}
	go a.run()
	defer func() { a.control <- shutdownMsg{}; <-a.done }()
	reviewRecv(t, stalled)
	sup := newSessionSupervisor(a.store.dataDir, a.supCfg, nil, nil)
	sup.resident[a.id] = &residentEntry{handle: &sessionHandle{inbox: a.inbox, control: a.control, done: a.done}}
	ws := newWSActor()
	server := newWSServer(a.store.dataDir, a.supCfg, sup, nil, nil, nil, ws)
	get, _ := json.Marshal(map[string]any{"type": "get_session", "sessionId": a.id})
	dispatched := make(chan struct{})
	go func() {
		// Same sequential calls as link.connectLoop's socket read loop.
		server.dispatch(get)
		server.dispatch([]byte(`{"type":"health"}`))
		close(dispatched)
	}()
	select {
	case <-dispatched:
	case <-time.After(150 * time.Millisecond):
		t.Error("a busy session blocks dispatch of an unrelated host-level health command")
	}
	close(release)
	reviewRecv(t, dispatched)
}

func TestV2ArchitectureReconnectApproval(t *testing.T) {
	a := reviewActor(t)
	a.state, a.gen, a.rec.Status = stateRunning, 1, "running"
	a.rec.Options = SessionOptions{Access: "ask", Mode: "build"}
	ws := newWSActor()
	a.wsSend = func(ev any) { ws.emit(ev) }
	sup := newSessionSupervisor(a.store.dataDir, a.supCfg, nil, nil)
	sup.resident[a.id] = &residentEntry{handle: &sessionHandle{inbox: a.inbox, control: a.control, done: a.done}}
	server := newWSServer(a.store.dataDir, a.supCfg, sup, nil, nil, nil, ws)
	go a.run()
	defer func() { a.control <- shutdownMsg{}; <-a.done }()
	a.inbox <- Envelope{Payload: workerApprovalReqMsg{gen: 1, id: "approval", callID: "approval", tool: "bash", reply: make(chan approvalOutcome, 1)}}
	if initial := reviewRecv(t, ws.outbound).(map[string]any); initial["type"] != "tool_approval_request" {
		t.Fatal(initial)
	}
	raw, _ := json.Marshal(map[string]any{"type": "get_session", "sessionId": a.id})
	server.dispatch(raw)
	response := reviewRecv(t, ws.outbound).(map[string]any)
	if response["type"] != "session_data" {
		t.Fatal(response)
	}
	payload := response["session"].(map[string]any)
	if payload["pendingApproval"] == nil {
		t.Error("reconnected browser receives no pendingApproval although the actor is waiting for that decision")
	}
}
